Advance `Dependencies/Luna-UI` to the committed Luna C2.5D SHA first.

```bash
cd ~/GitHub/Moth-Text
unzip -o ~/Downloads/Moth-Text-C2.5D-Repository-Overlay.zip -d .

git apply --recount --check patches/C2.5D_MOTH_FRAME_REPORTING.patch
git apply --recount patches/C2.5D_MOTH_FRAME_REPORTING.patch

git diff --check
swift package reset
swift package resolve
swift test --filter MothC25DFramePathDiagnosticsTests
swift build
swift test
./scripts/test-paired.sh
./scripts/validate-iteration.sh
./scripts/smoke-test-plugin-host.sh
```

```bash
swift run MothTextLinux
```

```bash
git add \
```
