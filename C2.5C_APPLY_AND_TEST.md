Advance `Dependencies/Luna-UI` to the committed Luna C2.5C SHA first.

```bash
cd ~/GitHub/Moth-Text
unzip -o ~/Downloads/Moth-Text-C2.5C-Repository-Overlay.zip -d .
git diff --check
swift package reset
swift package resolve
swift test --filter MothC25CVisualRowIndexStoreTests

swift test
./scripts/test-paired.sh
./scripts/validate-iteration.sh
./scripts/smoke-test-plugin-host.sh
```

